⚙ GASスクリプトへの追加コード ▼
既存のGASスクリプトの doPost 関数内の先頭付近に以下を追加してください。追加後は「デプロイを管理」→新バージョンで再デプロイしてください。 
// ===== 出欠データ取得 =====
if (d.action === 'getFormData') {
  try {
    var ss = SpreadsheetApp.openById(d.sid);
    var sheet;
    if (d.sheetName) {
      sheet = ss.getSheetByName(d.sheetName);
    } else if (d.gid) {
      var sheets = ss.getSheets();
      for (var i = 0; i < sheets.length; i++) {
        if (String(sheets[i].getSheetId()) === String(d.gid)) {
          sheet = sheets[i]; break;
        }
      }
    }
    if (!sheet) sheet = ss.getSheets()[0];
    var values = sheet.getDataRange().getValues();
    var tsv = values.map(function(r){
      return r.map(function(c){ return String(c); }).join('\t');
    }).join('\n');
    return ContentService
      .createTextOutput(JSON.stringify({ok:true,data:tsv}))
      .setMimeType(ContentService.MimeType.JSON);
  } catch(err) {
    return ContentService
      .createTextOutput(JSON.stringify({ok:false,error:err.message}))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
